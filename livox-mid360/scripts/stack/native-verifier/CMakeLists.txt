# 只构建实收验收器；不启动雷达，也不改动驱动或 FAST-LIO 源码。
cmake_minimum_required(VERSION 3.14)
project(livox_stack_native_verifier LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(verify_live_stream_native src/verify_live_stream.cpp)
target_compile_features(verify_live_stream_native PRIVATE cxx_std_17)
target_compile_options(verify_live_stream_native PRIVATE -O2 -Wall -Wextra)
ament_target_dependencies(verify_live_stream_native rclcpp livox_ros_driver2 sensor_msgs nav_msgs)
install(TARGETS verify_live_stream_native DESTINATION lib/${PROJECT_NAME})
ament_package()
